### 任务 3：实现词根匹配引擎

**文件：**

- [ ] **步骤 1：编写匹配算法函数**

```javascript
// 按类型和优先级排序的索引
const ROOTS_BY_TYPE = {
  root: ROOT_DB.filter(e => e.type === 'root').sort((a,b) => b.priority - a.priority),
  prefix: ROOT_DB.filter(e => e.type === 'prefix').sort((a,b) => b.priority - a.priority),
  suffix: ROOT_DB.filter(e => e.type === 'suffix').sort((a,b) => b.priority - a.priority)
};

function analyzeWord(word) {
  if (!word || word.trim().length === 0) return null;
  const w = word.trim().toLowerCase();
  let remaining = w;
  const parts = [];
  
  // 1. 前缀匹配 — 从开头匹配最长前缀
  let matchedPrefix = null;
  for (const p of ROOTS_BY_TYPE.prefix) {
    if (remaining.startsWith(p.root)) {
      matchedPrefix = { ...p, matchedText: p.root, start: 0, end: p.root.length };
      remaining = remaining.slice(p.root.length);
      break;
    }
  }
  if (matchedPrefix) parts.push(matchedPrefix);
  
  // 2. 词根匹配 — 从剩余部分找最长匹配
  let matchedRoot = null;
  let bestRoot = null;
  for (const r of ROOTS_BY_TYPE.root) {
    const idx = remaining.indexOf(r.root);
    if (idx !== -1) {
      if (!bestRoot || r.root.length > bestRoot.root.length) {
        bestRoot = { ...r, matchedText: r.root, start: matchedPrefix ? matchedPrefix.end + idx : idx, end: (matchedPrefix ? matchedPrefix.end + idx : idx) + r.root.length };
      }
    }
  }
  if (bestRoot) {
    matchedRoot = bestRoot;
    parts.push(matchedRoot);
  }
  
  // 3. 后缀匹配 — 从剩余尾部匹配最长后缀
  // 去掉已匹配的词根部分重新计算剩余
  remaining = w;
  if (matchedPrefix) remaining = remaining.slice(matchedPrefix.root.length);
  if (matchedRoot) {
    const rootIdx = remaining.indexOf(matchedRoot.root);
    if (rootIdx !== -1) {
      remaining = remaining.slice(rootIdx + matchedRoot.root.length);
    }
  }
  
  let matchedSuffix = null;
  for (const s of ROOTS_BY_TYPE.suffix) {
    if (remaining.endsWith(s.root)) {
      matchedSuffix = { ...s, matchedText: s.root, start: w.length - s.root.length, end: w.length };
      remaining = remaining.slice(0, remaining.length - s.root.length);
      break;
    }
  }
  if (matchedSuffix) parts.push(matchedSuffix);
  
  // 4. 未识别部分
  let unmatched = w;
  // 从剩余中算出未匹配部分
  const unrecognized = [];
  if (matchedPrefix) unmatched = unmatched.slice(matchedPrefix.root.length);
  if (matchedRoot) {
    const rootIdx = unmatched.indexOf(matchedRoot.root);
    if (rootIdx > 0) unrecognized.push({ text: unmatched.slice(0, rootIdx), type: 'unrecognized' });
    unmatched = unmatched.slice(rootIdx + matchedRoot.root.length);
  }
  if (matchedSuffix) {
    unmatched = unmatched.slice(0, unmatched.length - matchedSuffix.root.length);
  }
  if (unmatched.length > 0) unrecognized.push({ text: unmatched, type: 'unrecognized' });
  
  return { word: w, parts, unrecognized };
}
```

```javascript
function renderResults(result) {
  if (!result || result.parts.length === 0) {
    resultsEl.innerHTML = `
      <div class="no-match">
        <div class="no-match-icon">&#128269;</div>
        <p>未找到词根匹配</p>
        <p class="no-match-hint">试试启用 AI 增强模式，或检查单词拼写</p>
      </div>`;
    return;
  }
  
  let html = '<div class="cards-grid">';
  
  // 先展示未识别部分
  for (const u of result.unrecognized) {
    html += `
      <div class="root-card unrecognized">
        <div class="root-card-type">未识别</div>
        <div class="root-card-root">${u.text}</div>
        <div class="root-card-meaning">未匹配到词根</div>
        <div class="root-card-origin">—</div>
      </div>`;
  }
  
  // 展示匹配的每个部分
  for (const p of result.parts) {
    const typeLabel = p.type === 'prefix' ? '前缀' : p.type === 'suffix' ? '后缀' : '词根';
    const examplesHtml = p.examples.slice(0,5).map(ex => 
      `<span class="root-example" data-word="${ex}">${ex}</span>`
    ).join('');
    
    html += `
      <div class="root-card" data-type="${p.type}">
        <div class="root-card-type">${typeLabel}</div>
        <div class="root-card-root">${p.root}</div>
        <div class="root-card-meaning">${p.meaning}<br><span class="meaning-en">${p.meaning_en}</span></div>
        <div class="root-card-origin">${p.origin}${p.from ? ' · ' + p.from : ''}</div>
        ${examplesHtml ? `<div class="root-card-examples">${examplesHtml}</div>` : ''}
      </div>`;
  }
  
  html += '</div>';
  
  // 添加整体拼合说明
  html += '<div class="analysis-summary">';
  const partsStr = result.parts.map(p => `${p.root} (${p.meaning})`).join(' + ');
  html += `<p><strong>拆解：</strong> ${result.word} = ${partsStr}</p>`;
  html += '</div>';
  
  resultsEl.innerHTML = html;
}
```

```javascript
let aiEnabled = localStorage.getItem('root_ai_enabled') === 'true';
let apiKey = localStorage.getItem('root_api_key') || '';

function toggleAI() {
  aiEnabled = !aiEnabled;
  localStorage.setItem('root_ai_enabled', aiEnabled);
  document.getElementById('ai-toggle').classList.toggle('on', aiEnabled);
  // 如果有当前结果，重新分析
  const word = input.value.trim();
  if (word) {
    input.dispatchEvent(new Event('input'));
  }
}

function toggleApiPanel() {
  document.getElementById('api-key-panel').classList.toggle('show');
}

function saveApiKey() {
  apiKey = document.getElementById('api-key-input').value.trim();
  localStorage.setItem('root_api_key', apiKey);
  document.getElementById('api-key-panel').classList.remove('show');
}

async function analyzeWithAI(word, language) {
  if (!apiKey) {
    showToast('请先设置 API Key');
    return null;
  }
  
  const prompt = `分析单词 "${word}" 的词根词源结构。
语言：${language}
要求：
1. 拆解前缀/词根/后缀，每个部分给出含义
2. 给出词源来源（拉丁语/希腊语/古法语等）
3. 给出整体含义的演变逻辑
4. 对每个组成部分给出 2-3 个同根词示例
5. 以 JSON 格式返回，格式：{ "parts": [{"type":"prefix/root/suffix","root":"...","meaning":"...","origin":"...","examples":[...]}], "summary": "..." }`;

  try {
    const resp = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${apiKey}` },
      body: JSON.stringify({
        model: 'gpt-3.5-turbo',
        messages: [{ role: 'user', content: prompt }],
        temperature: 0.3,
        max_tokens: 1000
      })
    });
    const data = await resp.json();
    const content = data.choices?.[0]?.message?.content;
    if (content) {
      try {
        return JSON.parse(content);
      } catch {
        // 如果不是 JSON，返回原文
        return { raw: content };
      }
    }
  } catch (err) {
    console.error('AI analysis failed:', err);
  }
  return null;
}
```

```javascript
function renderAIResult(aiResult, localResult) {
  // 渲染本地结果作为基础
  renderResults(localResult);
  
  // 渲染 AI 摘要
  const aiSummary = document.getElementById('ai-summary');
  let html = '<div class="ai-section">';
  html += '<div class="ai-section-header">';
  html += '<span class="ai-badge">AI 增强</span>';
  html += '</div>';
  
  if (aiResult.parts) {
    html += '<div class="ai-parts">';
    for (const p of aiResult.parts) {
      html += `<div class="ai-part">
        <span class="ai-part-type">${p.type}</span>
        <strong>${p.root}</strong> — ${p.meaning}
        <span class="ai-part-origin">${p.origin || ''}</span>
      </div>`;
    }
    html += '</div>';
  }
  
  if (aiResult.summary) {
    html += `<div class="ai-summary-text">${aiResult.summary}</div>`;
  }
  
  if (aiResult.raw) {
    html += `<div class="ai-raw">${aiResult.raw}</div>`;
  }
  
  html += '</div>';
  aiSummary.innerHTML = html;
  aiSummary.style.display = 'block';
}
```
